// MAX 2 SESSION MEMORY ENGINE (UDT)

type SessDrawings
    box[] b_arr
    line[] l_arr

method push_b(SessDrawings[] hist, box b) =>
    if array.size(hist) > 0 and not na(b)
        array.push(array.get(hist, array.size(hist) - 1).b_arr, b)

method push_l(SessDrawings[] hist, line l) =>
    if array.size(hist) > 0 and not na(l)
        array.push(array.get(hist, array.size(hist) - 1).l_arr, l)

method new_sess(SessDrawings[] hist) =>
    array.push(hist, SessDrawings.new(array.new_box(), array.new_line()))
    if array.size(hist) > 2
        SessDrawings old = array.shift(hist)
        for b in old.b_arr
            box.delete(b)
        for l in old.l_arr
            line.delete(l)

// UPCOMING 15-MIN SESSION PREVIEW ENGINE
// ==========================================
f_next_start_time(string sess_time) =>
    int h = math.round(str.tonumber(str.substring(sess_time, 0, 2)))
    int m = math.round(str.tonumber(str.substring(sess_time, 2, 4)))
    int y = year(time, "America/New_York")
    int mo = month(time, "America/New_York")
    int d = dayofmonth(time, "America/New_York")
    
    int ts = timestamp("America/New_York", y, mo, d, h, m)
    if time >= ts
        ts += 86400000
        
    int day_of_wk = dayofweek(ts, "America/New_York")
    if day_of_wk == 7 
        if h >= 17
            ts += 86400000 
        else
            ts += 2 * 86400000 
    else if day_of_wk == 1 
        if h < 17
            ts += 86400000
    ts

var line preview_line = na
var box preview_box = na

if barstate.islast and kz_filter and kz_style == "Label"
    int next_ts = 99999999999999
    color next_color = na
    string next_text = ""
    
    if show_asia_kz
        int ts = f_next_start_time(asia_time_inp)
        if ts < next_ts
            next_ts := ts, next_color := asia_color, next_text := "asia"
    if show_lokz_kz
        int ts = f_next_start_time(lokz_time_inp)
        if ts < next_ts

    if show_nyam_kz

    if show_nylunch_kz

    if show_nypm_kz
        int ts = f_next_start_time(nypm_time_inp)
        if ts < next_ts
            next_ts := ts, next_color := nypm_color, next_text := "ny-pm"
            
    if (next_ts - time) <= 900000 and (next_ts > time)
        float box_top = high + (high-low) * 0.5
        float box_bot = low - (high-low) * 0.5
        float line_y_ext = vert_line_origin == "Top" ? low - (high-low)*3 : high + (high-low)*3
        float line_y_anchor = vert_line_origin == "Top" ? box_bot : box_top
        
        if na(preview_line)
            preview_line := line.new(next_ts, line_y_anchor, next_ts, line_y_ext, xloc=xloc.bar_time, color=color.new(next_color, kz_line_transp), style=vert_line_style)
            preview_box := box.new(next_ts, box_top, next_ts, box_bot, xloc=xloc.bar_time, border_color=color.new(color.white, 100), bgcolor=next_color, text=next_text, text_color=kz_text_color, text_size=kz_text_size)
        else
            line.set_x1(preview_line, next_ts)
            line.set_x2(preview_line, next_ts)
            line.set_y1(preview_line, line_y_anchor)
            line.set_y2(preview_line, line_y_ext)
            line.set_color(preview_line, color.new(next_color, kz_line_transp))
            
            box.set_left(preview_box, next_ts)
            box.set_right(preview_box, next_ts)
            box.set_top(preview_box, box_top)
            box.set_bottom(preview_box, box_bot)
            box.set_bgcolor(preview_box, next_color)
            box.set_text(preview_box, next_text)
    else
