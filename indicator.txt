// Tracking Objects with Timestamps for Clean Removal
type DrawingObj
    box b_obj
    line l_obj
    int time_created

var DrawingObj[] kz_drawings    = array.new<DrawingObj>()
var DrawingObj[] range_drawings = array.new<DrawingObj>()

// ==========================================
// 7. KILLZONE BOX & LABEL CALCULATIONS
// ==========================================
var float top_bottom = 0
var color kz_color   = na
var bool in_kz       = na
var string kz_text   = na

if tf_is_valid
    bool in_nyam_now = show_nyam_kz and not na(time(timeframe.period, nyam_time_inp, timezone = "America/New_York"))
    bool end_nyam_trigger = in_nyam_now[1] and not in_nyam_now

    // When NY AM closes, delete any drawings created more than 48 hours ago
    if end_nyam_trigger
        int purge_threshold = time - 172800000
        
        if array.size(kz_drawings) > 0
            for i = array.size(kz_drawings) - 1 to 0
                DrawingObj item = array.get(kz_drawings, i)
                if item.time_created < purge_threshold
                    if not na(item.b_obj)
                        box.delete(item.b_obj)
                    if not na(item.l_obj)
                        line.delete(item.l_obj)
                    array.remove(kz_drawings, i)

        if array.size(range_drawings) > 0
            for i = array.size(range_drawings) - 1 to 0
                DrawingObj item = array.get(range_drawings, i)
                if item.time_created < purge_threshold
                    if not na(item.b_obj)
                        box.delete(item.b_obj)
                    if not na(item.l_obj)
                        line.delete(item.l_obj)
                    array.remove(range_drawings, i)

    if kz_filter
        if show_asia_kz and not na(time(timeframe.period, asia_time_inp, timezone = "America/New_York"))
            in_kz := true
            kz_color := asia_color
            kz_text := "asia"
        else if show_lokz_kz and not na(time(timeframe.period, lokz_time_inp, timezone = "America/New_York"))
            in_kz := true
            kz_color := lokz_color
            kz_text := "lokz"
        else if in_nyam_now
            in_kz := true
            kz_color := nyam_color
            kz_text := "ny-am"
        else if show_nylunch_kz and not na(time(timeframe.period, nylunch_time_inp, timezone = "America/New_York"))
            in_kz := true
            kz_color := nylunch_color
            kz_text := "nyl" 
        else if show_nypm_kz and not na(time(timeframe.period, nypm_time_inp, timezone = "America/New_York"))
            in_kz := true
            kz_color := nypm_color
            kz_text := "ny-pm"
        else
            in_kz := false
    else
        in_kz := false

    var box curr_kz_box = na
    var line curr_kz_start_line = na
    var line curr_kz_end_line = na

    if kz_style == "Label"
        if in_kz and not in_kz[1]
            color_line = color.new(kz_color, kz_line_transp)
            curr_kz_box := box.new(bar_index, high, bar_index, low, border_color = color.new(color.white, 100), bgcolor = kz_color, text = kz_text, text_color = kz_text_color, text_size = kz_text_size)
            float box_y = vert_line_origin == "Top" ? box.get_top(curr_kz_box) : box.get_bottom(curr_kz_box)
            curr_kz_start_line := line.new(bar_index, high, bar_index, box_y, color = color_line, style = vert_line_style)
            
            array.push(kz_drawings, DrawingObj.new(curr_kz_box, na, time))
            array.push(kz_drawings, DrawingObj.new(na, curr_kz_start_line, time))
            top_bottom := 0

        if in_kz[1]
            box.set_right(curr_kz_box, bar_index)
            if high > box.get_bottom(curr_kz_box)
                box.set_bottom(curr_kz_box, 1.0010 * high)
                box.set_top(curr_kz_box, 1.0015 * high)
                float box_y = vert_line_origin == "Top" ? box.get_top(curr_kz_box) : box.get_bottom(curr_kz_box)
                line.set_y2(curr_kz_start_line, box_y)

        if in_kz[1] and not in_kz
            color_line = color.new(kz_color, kz_line_transp)
            float box_y = vert_line_origin == "Top" ? box.get_top(curr_kz_box) : box.get_bottom(curr_kz_box)
            curr_kz_end_line := line.new(bar_index, high, bar_index, box_y, color = color_line, style = vert_line_style)
            array.push(kz_drawings, DrawingObj.new(na, curr_kz_end_line, time))

    else if kz_style == "Top"
        top_bottom := 1
    else if kz_style == "Bottom"
        top_bottom := 2

    bool new_asia = in_asia and not in_asia[1]
    bool end_asia = in_asia[1] and not in_asia

    if new_asia and show_asia_box
        if range_type == "Wicks"
            asia_b := box.new(bar_index, high, bar_index, low, border_color = color.new(color.white, 100), bgcolor = asia_color)
        else if range_type == "Bodies"
            asia_b := box.new(bar_index, open, bar_index, open, border_color = color.new(color.white, 100), bgcolor = asia_color)
        array.push(range_drawings, DrawingObj.new(asia_b, na, time))

    if in_asia and show_asia_box and not na(asia_b)
        box.set_right(asia_b, bar_index + 1)
        if range_type == "Wicks"
            if high > box.get_top(asia_b)
                box.set_top(asia_b, high)
            if low < box.get_bottom(asia_b)
                box.set_bottom(asia_b, low)
        else if range_type == "Bodies"
            if math.max(open, close) > box.get_top(asia_b)
                box.set_top(asia_b, math.max(open, close))
            if math.min(open, close) < box.get_bottom(asia_b)
                box.set_bottom(asia_b, math.min(open, close))

    if end_asia and show_asia_box and not na(asia_b)
        asia_draw_h := true
        asia_h := line.new(box.get_left(asia_b), box.get_top(asia_b), bar_index, box.get_top(asia_b), color = asia_color)
        asia_draw_l := true
        asia_l := line.new(box.get_left(asia_b), box.get_bottom(asia_b), bar_index, box.get_bottom(asia_b), color = asia_color)
        array.push(range_drawings, DrawingObj.new(na, asia_h, time))
        array.push(range_drawings, DrawingObj.new(na, asia_l, time))
        
        if pipsFilter and ((box.get_top(asia_b) - box.get_bottom(asia_b)) * 10000 > maxPips)
            box.delete(asia_b), line.delete(asia_h), line.delete(asia_l)
            asia_draw_h := false, asia_draw_l := false
            
        if not ext_asia_line
            line.delete(asia_h), line.delete(asia_l)
            asia_draw_h := false, asia_draw_l := false

        line.set_x2(asia_h, bar_index)
        if high > line.get_y1(asia_h)
            asia_draw_h := false 

    if asia_draw_l and not na(asia_l)
        line.set_x2(asia_l, bar_index)
        if low < line.get_y1(asia_l)
            asia_draw_l := false

    var line lokz_h = na, var line lokz_l = na

            if math.min(open, close) < box.get_bottom(lokz_b)
                box.set_bottom(lokz_b, math.min(open, close))

        array.push(range_drawings, DrawingObj.new(na, lokz_h, time))

            line.delete(lokz_h), line.delete(lokz_l)

    if end_nyam and show_nyam_box and not na(nyam_b)
        nyam_draw_h := true
        nyam_h := line.new(box.get_left(nyam_b), box.get_top(nyam_b), bar_index, box.get_top(nyam_b), color = nyam_color)

    bool in_nyl = not na(time(timeframe.period, nylunch_time_inp, timezone="America/New_York"))
