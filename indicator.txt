// ==========================================
// FIXED GLOBAL 4H FUNCTION
// ==========================================
f_check_h4(start_sess, stop_hour, toggle, lbl_text) =>
    var line h4_line = na
    var label h4_lbl  = na
    var bool is_active = false
    
    bool is_start = not na(time(timeframe.period, session = start_sess, timezone = "America/New_York"))
    bool within_3_days = (timenow - time) <= 259200000
    
    if is_start and not is_start[1] and show_4h_opens and toggle and within_3_days
        is_active := true
        h4_line := line.new(bar_index, open, bar_index + 1, open, color = h4_color, style = h4_style, width = h4_width)
        if show_h4_labels
            h4_lbl := label.new(bar_index + 1, open, lbl_text, color = color.new(color.white, 100), textcolor = h4_color, style = label.style_label_left, size = size.tiny)
            
    else if is_active and not na(h4_line)
        current_hour = hour(time, "America/New_York")
        bool should_stop = false
        
        if stop_hour == 17
            if current_hour >= 17 or current_hour < 6
                should_stop := true
        if stop_hour == 6
            if current_hour >= 6 and current_hour < 17
                should_stop := true
                
        if should_stop
            is_active := false 
        else
            line.set_x2(h4_line, bar_index + 1)
            if show_h4_labels
                label.set_x(h4_lbl, bar_index + 1)
                
    int(na)

            kz_text := "lokz"

            kz_text := "ny-am"

            kz_text := "nyl"

    var line kz_end_line = na

            color_line = color.new(kz_color, kz_line_transp)
            kz_box := box.new(bar_index, high, bar_index, low, border_color = color.new(color.white, 100), bgcolor = kz_color, text = kz_text, text_color = kz_text_color, text_size = kz_text_size)
            float box_y = vert_line_origin == "Top" ? box.get_top(kz_box) : box.get_bottom(kz_box)
            kz_start_line := line.new(bar_index, high, bar_index, box_y, color = color_line, style = vert_line_style)
            top_bottom := 0
        if in_kz[1]
            box.set_right(kz_box, bar_index)
            if high > box.get_bottom(kz_box)
                box.set_bottom(kz_box, 1.0010 * high)
                box.set_top(kz_box, 1.0015 * high)
                float box_y = vert_line_origin == "Top" ? box.get_top(kz_box) : box.get_bottom(kz_box)
                line.set_y2(kz_start_line, box_y)
        if in_kz[1] and not in_kz
            color_line = color.new(kz_color, kz_line_transp)
            float box_y = vert_line_origin == "Top" ? box.get_top(kz_box) : box.get_bottom(kz_box)
            kz_end_line := line.new(bar_index, high, bar_index, box_y, color = color_line, style = vert_line_style)
    else if kz_style == "Top"
        top_bottom := 1
    else if kz_style == "Bottom"
        top_bottom := 2

// --- ASIA ---
var box asia_b = na
var line asia_h = na
var line asia_l = na
var bool asia_draw_h = false
var bool asia_draw_l = false

bool in_asia = not na(time(timeframe.period, asia_time_inp, timezone="America/New_York"))
bool new_asia = in_asia and not in_asia[1]
bool end_asia = in_asia[1] and not in_asia

if new_asia and show_asia_box
    if range_type == "Wicks"
        asia_b := box.new(bar_index, high, bar_index, low, border_color = color.new(color.white, 100), bgcolor = asia_color)
    else if range_type == "Bodies"
        asia_b := box.new(bar_index, open, bar_index, open, border_color = color.new(color.white, 100), bgcolor = asia_color)

if in_asia and show_asia_box and not na(asia_b)
    box.set_right(asia_b, bar_index + 1)
    if range_type == "Wicks"
        if high > box.get_top(asia_b)
            box.set_top(asia_b, high)
        if low < box.get_bottom(asia_b)
            box.set_bottom(asia_b, low)
    else if range_type == "Bodies"
        if math.max(open, close) > box.get_top(asia_b)
            box.set_top(asia_b, math.max(open, close))
        if math.min(open, close) < box.get_bottom(asia_b)
            box.set_bottom(asia_b, math.min(open, close))

if end_asia and show_asia_box and not na(asia_b)
    asia_draw_h := true
    asia_h := line.new(box.get_left(asia_b), box.get_top(asia_b), bar_index, box.get_top(asia_b), color = asia_color)
    asia_draw_l := true
    asia_l := line.new(box.get_left(asia_b), box.get_bottom(asia_b), bar_index, box.get_bottom(asia_b), color = asia_color)
    
    if pipsFilter and ((box.get_top(asia_b) - box.get_bottom(asia_b)) * 10000 > maxPips)
        box.delete(asia_b)
        line.delete(asia_h)
        line.delete(asia_l)
        asia_draw_h := false
        asia_draw_l := false
        
    if not ext_asia_line
        line.delete(asia_h)
        line.delete(asia_l)
        asia_draw_h := false
        asia_draw_l := false

if asia_draw_h and not na(asia_h)
    line.set_x2(asia_h, bar_index)
    if high > line.get_y1(asia_h)
        asia_draw_h := false 

if asia_draw_l and not na(asia_l)
    line.set_x2(asia_l, bar_index)
    if low < line.get_y1(asia_l)
        asia_draw_l := false
