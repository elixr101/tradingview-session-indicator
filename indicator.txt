// =============================================================================
// UNIFIED TIME-BASED SWEEP PRUNING, HUD & FIFO QUEUE ENGINE
// =============================================================================
type CurrentSweepState
    bool swept
    int  t

var CurrentSweepState sweep_state = CurrentSweepState.new(false, 0)
sweep_state.swept := false

type TimedSweep
    string id
    label lbl
    int t
    int max_ms

var TimedSweep[] intra_sweeps = array.new<TimedSweep>()
var TimedSweep[] htf_sweeps = array.new<TimedSweep>()

// Tracking Arrays for FIFO Deletion Limits
var label[] track_req  = array.new_label()
var label[] track_kz_h = array.new_label()
var label[] track_kz_l = array.new_label()
var label[] track_pdh  = array.new_label()
var label[] track_pdl  = array.new_label()
var label[] track_pwh  = array.new_label()
var label[] track_pwl  = array.new_label()
var label[] track_tdo  = array.new_label()
var label[] track_4h   = array.new_label()
var label[] track_gap  = array.new_label()

// HUD Arrays
var string[] alert_msgs = array.new_string()
var int[] alert_times = array.new_int()

// Master Sweep Creation
create_sweep_label(bool is_high, string cat, string alert_msg) =>
    label l = na
    if show_sweeps and (sweep_mode == "Both ('x' and Candle Color)" or sweep_mode == "Only 'x'")
        l := label.new(bar_index, is_high ? high : low, "x", color=color.new(color.white, 100), textcolor=sweep_color, style=is_high ? label.style_label_down : label.style_label_up, size=get_text_size(sweep_sz_str))
        
        // Dynamic FIFO Limit Enforcement
        if cat == "REQ"
            array.push(track_req, l)
            if array.size(track_req) > max_req_sweeps
                label.delete(array.shift(track_req))
        else if cat == "KZ_H"
            array.push(track_kz_h, l)
            if array.size(track_kz_h) > max_kz_sweeps
                label.delete(array.shift(track_kz_h))
        else if cat == "KZ_L"
            array.push(track_kz_l, l)
            if array.size(track_kz_l) > max_kz_sweeps
                label.delete(array.shift(track_kz_l))
        else if cat == "PDH"
            array.push(track_pdh, l)
            if array.size(track_pdh) > max_htf_sweeps
                label.delete(array.shift(track_pdh))
        else if cat == "PDL"
            array.push(track_pdl, l)
            if array.size(track_pdl) > max_htf_sweeps
                label.delete(array.shift(track_pdl))
        else if cat == "PWH"
            array.push(track_pwh, l)
            if array.size(track_pwh) > max_htf_sweeps
                label.delete(array.shift(track_pwh))
        else if cat == "PWL"
            array.push(track_pwl, l)
            if array.size(track_pwl) > max_htf_sweeps
                label.delete(array.shift(track_pwl))
        else if cat == "TDO"
            array.push(track_tdo, l)
            if array.size(track_tdo) > max_htf_sweeps
                label.delete(array.shift(track_tdo))
        else if cat == "4H"
            array.push(track_4h, l)
            if array.size(track_4h) > max_htf_sweeps
                label.delete(array.shift(track_4h))
        else if cat == "GAP"
            array.push(track_gap, l)
            if array.size(track_gap) > max_htf_sweeps
                label.delete(array.shift(track_gap))

    // Skip permanent barcolor baking ONLY for 4H (we will use a fake highlight instead)
    if cat != "4H"
        sweep_state.swept := true
        sweep_state.t := time
        
    // Generate HUD Alert
    if show_alert_hud
        bool is_duplicate = false
        if array.size(alert_msgs) > 0
            for i = 0 to array.size(alert_msgs) - 1
                if array.get(alert_msgs, i) == (alert_msg + " SWEPT") and array.get(alert_times, i) == time
                    is_duplicate := true
                    break
        if not is_duplicate
            array.unshift(alert_msgs, alert_msg + " SWEPT")
            array.unshift(alert_times, time)
    l

push_intra_sweep(label lbl) =>
    if not na(lbl)
        array.push(intra_sweeps, TimedSweep.new("intra", lbl, time, 172800000))

push_htf_sweep(string id, label lbl, int max_ms) =>
    if not na(lbl)
        array.push(htf_sweeps, TimedSweep.new(id, lbl, time, max_ms))

// Routine Pruning for Sweeps strictly by age
if array.size(intra_sweeps) > 0
    for i = array.size(intra_sweeps) - 1 to 0
        TimedSweep ts = array.get(intra_sweeps, i)
        if (time - ts.t) > ts.max_ms
            label.delete(ts.lbl)
            array.remove(intra_sweeps, i)

        array.push(array.get(hist, array.size(hist) - 1).l_arr, l)

method push_lbl(SessDrawings[] hist, label lbl) =>
    if array.size(hist) > 0 and not na(lbl)
        array.push(array.get(hist, array.size(hist) - 1).lbl_arr, lbl)

method new_sess(SessDrawings[] hist) =>
    array.push(hist, SessDrawings.new(array.new_box(), array.new_line(), array.new_label()))
    if array.size(hist) > 2
        SessDrawings old = array.shift(hist)
        for b in old.b_arr
            box.delete(b)
        for l in old.l_arr
            line.delete(l)
        for lbl in old.lbl_arr
            label.delete(lbl)

f_next_start_time(string sess_time) =>
    int h = math.round(str.tonumber(str.substring(sess_time, 0, 2)))
    int m = math.round(str.tonumber(str.substring(sess_time, 2, 4)))
    int y = year(time, "America/New_York")
    int mo = month(time, "America/New_York")
    int d = dayofmonth(time, "America/New_York")
    
    int ts = timestamp("America/New_York", y, mo, d, h, m)
    if time >= ts
        ts += 86400000
        
    int day_of_wk = dayofweek(ts, "America/New_York")
    if day_of_wk == 7 
        if h >= 17
            ts += 86400000 
        else
            ts += 2 * 86400000 
    else if day_of_wk == 1 
        if h < 17
            ts += 86400000
    ts

// 12. RELATIVE EQUAL HIGHS & LOWS (REQH / REQL) DIRECT WICK-TO-WICK ENGINE
// ==========================================
float dyn_eq_tolerance = eq_tol_ltf
int dyn_max_age = 172800000 // 2 Days default for LTF

if timeframe.isdaily or timeframe.isweekly or timeframe.ismonthly
    dyn_eq_tolerance := eq_tol_dtf
    dyn_max_age := 2592000000 // 1 Month
else if timeframe.isintraday and tf_curr_min >= 15
    dyn_eq_tolerance := eq_tol_htf
    dyn_max_age := 604800000 // 1 Week lifespan

int lookback_bars = math.max(10, math.round(eq_lookback_mins / math.max(tf_curr_min, 1)))

type EQLineWick
    line l
    label lbl
    float price_sweep
    bool is_high
    int start_time
    int end_bar
    bool is_swept
    int kz_id // Tracks which Killzone originated the level

var EQLineWick[] req_array = array.new<EQLineWick>()

float ph = ta.pivothigh(high, 2, 2)
float pl = ta.pivotlow(low, 2, 2)

if tf_is_valid
    // Detect REQH
    if show_reqh and not na(ph)
        int match_idx = na
        float match_val = na
        
        for i = 4 to lookback_bars
            if bar_index - i >= 0
                float past_h = high[i]
                bool is_local_peak = high[i] >= high[i+1] and high[i] >= high[i-1]
                
                if not na(past_h) and is_local_peak and math.abs(ph - past_h) <= dyn_eq_tolerance
                    bool intermediate_filled = false
                    for j = 3 to i - 1
                        if high[j] > math.max(ph, past_h)
                            intermediate_filled := true
                            break
                    
                    if not intermediate_filled
                        match_idx := bar_index - i
                        match_val := past_h
                        break

        bool level_exists = false
        if array.size(req_array) > 0
            for k = 0 to array.size(req_array) - 1
                EQLineWick active_eq = array.get(req_array, k)
                if active_eq.is_high and not active_eq.is_swept and math.abs(active_eq.price_sweep - ph) <= dyn_eq_tolerance
                    level_exists := true
                    break
        
        bool allow_creation = true
        if timeframe.isintraday and tf_curr_min < 15 and not in_kz
            allow_creation := false

        if not na(match_idx) and not level_exists and allow_creation
            int x1 = match_idx
            int x2 = bar_index - 2
            float y1 = match_val
            float y2 = ph
            float sweep_target = math.max(y1, y2)
            
            line l_diag = line.new(x1, y1, x2, y2, color=color.new(reqh_color, eq_transp), style=eq_line_style, width=eq_line_width)
            int lbl_x = eq_halign_str == "Center" ? math.round((x1 + x2) / 2) : (eq_halign_str == "Right" ? x2 : x1)
            float lbl_y = eq_valign_str == "Middle" ? (y1 + y2) / 2 : (eq_valign_str == "Bottom" ? math.min(y1, y2) : math.max(y1, y2))
            label lbl_eq = label.new(lbl_x, lbl_y, "reqh", color=color.new(color.white, 100), textcolor=eq_txt_color, style=get_eq_label_style(eq_valign_str), size=eq_text_size)
            
            int current_kz = active_kz_idx
            array.push(req_array, EQLineWick.new(l_diag, lbl_eq, sweep_target, true, time, x2, false, current_kz))
            
            if timeframe.isintraday and tf_curr_min < 15
                if current_kz == 1
                    asia_hist.push_l(l_diag), asia_hist.push_lbl(lbl_eq)
                else if current_kz == 2
                    lokz_hist.push_l(l_diag), lokz_hist.push_lbl(lbl_eq)
                else if current_kz == 3
                    nyam_hist.push_l(l_diag), nyam_hist.push_lbl(lbl_eq)
                else if current_kz == 4
                    nyl_hist.push_l(l_diag), nyl_hist.push_lbl(lbl_eq)
                else if current_kz == 5
                    nypm_hist.push_l(l_diag), nypm_hist.push_lbl(lbl_eq)

    // Detect REQL
    if show_reql and not na(pl)
        int match_l_idx = na
        float match_l_val = na
        
        for i = 4 to lookback_bars
            if bar_index - i >= 0
                float past_l = low[i]
                bool is_local_valley = low[i] <= low[i+1] and low[i] <= low[i-1]
                
                if not na(past_l) and is_local_valley and math.abs(pl - past_l) <= dyn_eq_tolerance
                    bool intermediate_filled = false
                    for j = 3 to i - 1
                        if low[j] < math.min(pl, past_l)
                            intermediate_filled := true
                            break
                    
                    if not intermediate_filled
                        match_l_idx := bar_index - i
                        match_l_val := past_l
                        break

        bool level_l_exists = false
        if array.size(req_array) > 0
            for k = 0 to array.size(req_array) - 1
                EQLineWick active_eq = array.get(req_array, k)
                if not active_eq.is_high and not active_eq.is_swept and math.abs(active_eq.price_sweep - pl) <= dyn_eq_tolerance
                    level_l_exists := true
                    break

        bool allow_creation = true
        if timeframe.isintraday and tf_curr_min < 15 and not in_kz
            allow_creation := false

        if not na(match_l_idx) and not level_l_exists and allow_creation
            int x1 = match_l_idx
            int x2 = bar_index - 2
            float y1 = match_l_val
            float y2 = pl
            float sweep_target = math.min(y1, y2)
            
            line l_diag = line.new(x1, y1, x2, y2, color=color.new(reql_color, eq_transp), style=eq_line_style, width=eq_line_width)
            int lbl_x = eq_halign_str == "Center" ? math.round((x1 + x2) / 2) : (eq_halign_str == "Right" ? x2 : x1)
            float lbl_y = eq_valign_str == "Middle" ? (y1 + y2) / 2 : (eq_valign_str == "Top" ? math.max(y1, y2) : math.min(y1, y2))
            label lbl_eq = label.new(lbl_x, lbl_y, "reql", color=color.new(color.white, 100), textcolor=eq_txt_color, style=get_eq_label_style(eq_valign_str), size=eq_text_size)
            
            int current_kz = active_kz_idx
            array.push(req_array, EQLineWick.new(l_diag, lbl_eq, sweep_target, false, time, x2, false, current_kz))
            
            if timeframe.isintraday and tf_curr_min < 15
                if current_kz == 1
                    asia_hist.push_l(l_diag), asia_hist.push_lbl(lbl_eq)
                else if current_kz == 2
                    lokz_hist.push_l(l_diag), lokz_hist.push_lbl(lbl_eq)
                else if current_kz == 3
                    nyam_hist.push_l(l_diag), nyam_hist.push_lbl(lbl_eq)
                else if current_kz == 4
                    nyl_hist.push_l(l_diag), nyl_hist.push_lbl(lbl_eq)
                else if current_kz == 5
                    nypm_hist.push_l(l_diag), nypm_hist.push_lbl(lbl_eq)

    // Sweep management & Pruning
    if array.size(req_array) > 0
        for i = array.size(req_array) - 1 to 0
            EQLineWick eq = array.get(req_array, i)
            
            if (time - eq.start_time) > dyn_max_age
                line.delete(eq.l)
                label.delete(eq.lbl)
                array.remove(req_array, i)
            else if not eq.is_swept and bar_index > eq.end_bar
                if eq.is_high and high >= eq.price_sweep
                    eq.is_swept := true
                    label l = create_sweep_label(true, "REQ", "REQH")
                    
                    if timeframe.isintraday and tf_curr_min < 15
                        if eq.kz_id == 1
                            asia_hist.push_lbl(l)
                        else if eq.kz_id == 2
                            lokz_hist.push_lbl(l)
                        else if eq.kz_id == 3
                            nyam_hist.push_lbl(l)
                        else if eq.kz_id == 4
                            nyl_hist.push_lbl(l)
                        else if eq.kz_id == 5
                            nypm_hist.push_lbl(l)
                        else
                            push_intra_sweep(l)
                    else
                        push_htf_sweep("REQH_SWP", l, dyn_max_age)
                        
                    line.delete(eq.l)
                    label.delete(eq.lbl)
                    array.remove(req_array, i)
                        
                else if not eq.is_high and low <= eq.price_sweep
                    eq.is_swept := true
                    label l = create_sweep_label(false, "REQ", "REQL")
                    
                    if timeframe.isintraday and tf_curr_min < 15
                        if eq.kz_id == 1
                            asia_hist.push_lbl(l)
                        else if eq.kz_id == 2
                            lokz_hist.push_lbl(l)
                        else if eq.kz_id == 3
                            nyam_hist.push_lbl(l)
                        else if eq.kz_id == 4
                            nyl_hist.push_lbl(l)
                        else if eq.kz_id == 5
                            nypm_hist.push_lbl(l)
                        else
                            push_intra_sweep(l)
                    else
                        push_htf_sweep("REQL_SWP", l, dyn_max_age)
                        
                    line.delete(eq.l)
                    label.delete(eq.lbl)
                    array.remove(req_array, i)
